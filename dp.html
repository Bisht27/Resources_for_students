📚 Problem: Fibonacci Sequence

The Fibonacci sequence is defined as: F(0) = 0, F(1) = 1, F(n) = F(n-1) + F(n-2) for n ≥ 2.

This is a classic example to understand the power of Dynamic Programming!

⚠️ Warning

Values above 35 may take a long time for brute-force recursion!

Approach Time Complexity Space Complexity Notes
Brute Force (Recursion) O(φⁿ) ≈ O(2ⁿ) O(n) Exponential time, recomputes subproblems
Memoization (Top-Down DP) O(n) O(n) Stores results, avoids recomputation
Tabulation (Bottom-Up DP) O(n) O(n) Iterative, builds solution from base cases
Space Optimized O(n) O(1) Only stores last two values

💻 Implementation Code

// 1. Brute Force Recursion
function fibRecursive(n) {
    if (n <= 1) return n;
    return fibRecursive(n - 1) + fibRecursive(n - 2);
}

// 2. Memoization (Top-Down DP)
function fibMemo(n, memo = {}) {
    if (n <= 1) return n;
    if (memo[n]) return memo[n];
    memo[n] = fibMemo(n - 1, memo) + fibMemo(n - 2, memo);
    return memo[n];
}

// 3. Tabulation (Bottom-Up DP)
function fibDP(n) {
    if (n <= 1) return n;
    const dp = new Array(n + 1);
    dp[0] = 0;
    dp[1] = 1;
    for (let i = 2; i <= n; i++) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }
    return dp[n];
}

// 4. Space Optimized
function fibOptimized(n) {
    if (n <= 1) return n;
    let a = 0, b = 1;
    for (let i = 2; i <= n; i++) {
        [a, b] = [b, a + b];
    }
    return b;
}

📚 Problem: 0/1 Knapsack

Given items with weights and values, and a knapsack with capacity W, find the maximum value you can carry.

Each item can be taken at most once (0/1 choice).

Approach Time Complexity Space Complexity
Brute Force O(2ⁿ) O(n)
DP (2D Table) O(n × W) O(n × W)
Space Optimized DP O(n × W) O(W)

💻 Implementation Code

// Bottom-Up DP Solution
function knapsack(weights, values, capacity) {
    const n = weights.length;
    const dp = Array(n + 1).fill(null)
        .map(() => Array(capacity + 1).fill(0));
    
    for (let i = 1; i <= n; i++) {
        for (let w = 0; w <= capacity; w++) {
            if (weights[i - 1] <= w) {
                dp[i][w] = Math.max(
                    dp[i - 1][w],
                    values[i - 1] + dp[i - 1][w - weights[i - 1]]
                );
            } else {
                dp[i][w] = dp[i - 1][w];
            }
        }
    }
    return dp[n][capacity];
}

📚 Problem: Longest Common Subsequence

Find the length of the longest subsequence common to two sequences.

Example: LCS of "ABCDGH" and "AEDFHR" is "ADH" with length 3.

Approach Time Complexity Space Complexity
Brute Force O(2^(m+n)) O(m + n)
DP (2D Table) O(m × n) O(m × n)

💻 Implementation Code

// Bottom-Up DP Solution
function lcs(str1, str2) {
    const m = str1.length;
    const n = str2.length;
    const dp = Array(m + 1).fill(null)
        .map(() => Array(n + 1).fill(0));
    
    for (let i = 1; i <= m; i++) {
        for (let j = 1; j <= n; j++) {
            if (str1[i - 1] === str2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
    }
    return dp[m][n];
}

📚 Problem: Coin Change

Given coins of different denominations and a total amount, find the minimum number of coins needed to make that amount.

If the amount cannot be made up, return -1.

Approach Time Complexity Space Complexity
Brute Force O(S^n) - exponential O(n)
DP (Bottom-Up) O(amount × coins) O(amount)

💻 Implementation Code

// Bottom-Up DP Solution
function coinChange(coins, amount) {
    const dp = Array(amount + 1).fill(Infinity);
    dp[0] = 0;
    
    for (let i = 1; i <= amount; i++) {
        for (let coin of coins) {
            if (i >= coin) {
                dp[i] = Math.min(dp[i], dp[i - coin] + 1);
            }
        }
    }
    
    return dp[amount] === Infinity ? -1 : dp[amount];
}
